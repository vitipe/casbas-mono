# Tokenizador mínimo: 0O 1lI son distinguibles
from dataclasses import dataclass

@dataclass
class Token:
    kind: str
    value: str | None = None

def tokenize(src: str) -> list[Token]:
    out, i = [], 0
    while i < len(src):
        c = src[i]
        if c.isdigit():
            j = i
            while j < len(src) and src[j].isdigit(): j += 1
            out.append(Token("num", src[i:j])); i = j
        elif c in "+-*/=!<>":
            out.append(Token("op", c)); i += 1
        else:
            i += 1  # ignorar espacios: ¿y los acentos? ¡también!
    return out

/// Cuenta palabras ignorando mayúsculas
fn count(text: &str) -> HashMap<String, usize> {
    let mut map = HashMap::new();
    for w in text.split_whitespace() {
        *map.entry(w.to_lowercase()).or_insert(0) += 1;
    }
    map
}
